cmake_minimum_required(VERSION 3.21)
project(WeatherClockConsole LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network)
qt_add_executable(WeatherClockConsole main.cpp device.cpp device.h discovery.cpp discovery.h)
target_link_libraries(WeatherClockConsole PRIVATE Qt6::Widgets Qt6::Network advapi32)
target_compile_definitions(WeatherClockConsole PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(WeatherClockConsole PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../../stm32/ota")
target_compile_options(WeatherClockConsole PRIVATE -finput-charset=UTF-8 -fexec-charset=UTF-8)
